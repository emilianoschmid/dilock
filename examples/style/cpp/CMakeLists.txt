cmake_minimum_required(VERSION 3.20)
project(dilock_style_cpp LANGUAGES CXX)
find_package(Threads REQUIRED)
add_executable(dilock_style main.cpp)
target_compile_features(dilock_style PRIVATE cxx_std_20)
target_link_libraries(dilock_style PRIVATE Threads::Threads)
if(MSVC)
  target_compile_options(dilock_style PRIVATE /W4 /UNDEBUG)
else()
  target_compile_options(dilock_style PRIVATE -Wall -Wextra -Wpedantic -UNDEBUG)
endif()
enable_testing()
add_test(NAME style COMMAND dilock_style)
